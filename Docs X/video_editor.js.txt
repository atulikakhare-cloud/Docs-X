function initVideoEditorUI() {
  const container = document.getElementById('videoEditorView');
  if (!container) return;

  container.innerHTML = `
    <div style="background: #181825; padding: 20px; border-radius: 8px; border: 1px solid #313244; display: flex; flex-direction: column; gap: 15px; align-items: center; justify-content: center; flex: 1;">
      <input type="file" id="videoUploadInput" accept="video/*" style="display: none;" onchange="handleVideoUpload(event)">
      <button onclick="document.getElementById('videoUploadInput').click()" style="background: #89b4fa; color: #11111b; padding: 12px 24px; font-size: 15px;">🎬 Upload Video</button>
      <div id="videoContainer" style="width: 100%; height: 350px; background: #11111b; border: 1px solid #45475a; border-radius: 6px; display: flex; align-items: center; justify-content: center; overflow: hidden; position: relative;">
        <p style="color: #a6adc8;">No video loaded yet</p>
      </div>
    </div>
  `;
}

function handleVideoUpload(e) {
  const file = e.target.files[0];
  if (!file) return;

  const videoUrl = URL.createObjectURL(file);
  const container = document.getElementById('videoContainer');
  container.innerHTML = `
    <video controls style="max-width: 100%; max-height: 100%;">
      <source src="${videoUrl}" type="${file.type}">
      Your browser does not support the video tag.
    </video>
  `;
}
