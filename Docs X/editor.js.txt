const { ipcRenderer } = require('electron');

document.addEventListener('DOMContentLoaded', () => {
  const editor = document.getElementById('editor');
  const charCount = document.getElementById('charCount');

  if (editor && charCount) {
    editor.addEventListener('input', () => {
      charCount.innerText = `${editor.value.length} characters`;
    });
  }
});

async function saveTXT() {
  const editor = document.getElementById('editor');
  const text = editor ? editor.value : '';
  if (!text.trim()) return alert('Please write something first!');
  
  const res = await ipcRenderer.invoke('save-file', {
    content: text,
    defaultName: 'document.txt',
    filters: [{ name: 'Text File', extensions: ['txt'] }]
  });

  if (res.success) alert('File saved as .txt successfully!');
}

async function savePDF() {
  const editor = document.getElementById('editor');
  const text = editor ? editor.value : '';
  if (!text.trim()) return alert('Please write something first!');

  const { jsPDF } = window.jspdf;
  const doc = new jsPDF();
  
  const splitText = doc.splitTextToSize(text, 180);
  doc.text(splitText, 10, 10);

  const pdfData = doc.output('arraybuffer');
  const buffer = Buffer.from(pdfData);

  const res = await ipcRenderer.invoke('save-file', {
    content: buffer,
    defaultName: 'document.pdf',
    filters: [{ name: 'PDF Document', extensions: ['pdf'] }]
  });

  if (res.success) alert('File exported as .pdf successfully!');
}

async function saveDOCX() {
  const editor = document.getElementById('editor');
  const text = editor ? editor.value : '';
  if (!text.trim()) return alert('Please write something first!');

  const docxLib = window.docx;
  const doc = new docxLib.Document({
    sections: [{
      children: text.split('\n').map(line => new docxLib.Paragraph({
        children: [new docxLib.TextRun(line)]
      }))
    }]
  });

  const blob = await docxLib.Packer.toBuffer(doc);

  const res = await ipcRenderer.invoke('save-file', {
    content: blob,
    defaultName: 'document.docx',
    filters: [{ name: 'Word Document', extensions: ['docx'] }]
  });

  if (res.success) alert('File exported as .docx successfully!');
}
